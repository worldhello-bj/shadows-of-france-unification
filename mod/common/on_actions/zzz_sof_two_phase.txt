 on_capitulation = { effect = { if = { limit = { has_capitulated = yes } sof_tp_cancel_project_effect = yes if = { limit = { has_country_flag = sof_tp_contract_active } sof_tp_clear_contract = yes } } FROM = { if = { limit = { has_capitulated = yes } sof_tp_cancel_project_effect = yes if = { limit = { has_country_flag = sof_tp_contract_active } sof_tp_clear_contract = yes } } } } }
