sofzh_chart_party_liberal = {
 OR = {
  AND = { has_government = democratic OR = { has_country_leader_ideology = liberalism has_country_leader_ideology = populism } }
  AND = { NOT = { has_government = democratic } OR = {
    any_character = { OR = { has_ideology = liberalism has_ideology = populism } PREV = { has_country_leader = { character = PREV ruling_only = no } } }
  } }
 }
}
