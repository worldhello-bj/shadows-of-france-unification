# Restore the naval portion excluded by the 2.11 US land/air strategy.
sofzh_coastal_naval_production = {
    allowed = { always = yes }
    enable = {
        is_ai = yes
        capital_scope = { sofzh_logistics_french_state = yes }
        any_controlled_state = { is_coastal = yes }
    }
    abort_when_not_enabled = yes
    ai_strategy = { type = unit_ratio id = capital_ship value = 10 }
    ai_strategy = { type = unit_ratio id = submarine value = 10 }
    ai_strategy = { type = unit_ratio id = screen_ship value = 40 }
    ai_strategy = { type = unit_ratio id = convoy value = 15 }
}
sofzh_coastal_convoy_production = {
    allowed = { always = yes }
    enable = {

        any_controlled_state = { is_coastal = yes }
        num_of_naval_factories > 0

    abort_when_not_enabled = yes
    ai_strategy = { type = equipment_production_min_factories id = convoy value = 1 }
}
