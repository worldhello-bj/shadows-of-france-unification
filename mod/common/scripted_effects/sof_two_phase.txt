sof_tp_refresh = {
 if = { limit = { is_in_faction = no } set_variable = { sof_tp_bloc = 0 } }

 set_variable = { sof_tp_states = 0 } set_variable = { sof_tp_bloc_states = 0 }
 set_variable = { sof_tp_members_count = 0 } set_variable = { sof_tp_neutrals_count = 0 }
 every_owned_state = { limit = { sofzh_unification_french_state = yes is_fully_controlled_by = PREV } PREV = { add_to_variable = { sof_tp_states = 1 } } }
 set_variable = { sof_tp_bloc_states = sof_tp_states }
 every_other_country = {
  limit = { sof_tp_country = yes }
  if = { limit = { is_in_faction_with = PREV }
   if = { limit = { is_faction_leader = yes } PREV = { set_variable = { sof_tp_bloc = PREV.sof_tp_bloc } } }
   PREV = { add_to_array = { array = sof_tp_members value = PREV } add_to_variable = { sof_tp_members_count = 1 } }
   every_owned_state = { limit = { sofzh_unification_french_state = yes is_fully_controlled_by = PREV } PREV.PREV = { add_to_variable = { sof_tp_bloc_states = 1 } } }
  }
  else_if = { limit = { is_in_faction = no } PREV = { add_to_array = { array = sof_tp_neutrals value = PREV } add_to_variable = { sof_tp_neutrals_count = 1 } } }
  if = { limit = { has_country_flag = sof_tp_congress@PREV } PREV = { add_to_array = { array = sof_tp_participants value = PREV } } }
 }
 clamp_variable = { var = sof_tp_legitimacy min = 0 max = 100 }
 clamp_variable = { var = sof_tp_trust min = 0 max = 100 }
 clamp_variable = { var = sof_tp_reconstruction min = 0 max = 100 }

}
