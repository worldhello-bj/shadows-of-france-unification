 set_variable = { var = sofzh_merge_remaining value = 0 }
 every_country = {
  limit = { exists = yes }
  ROOT = { add_to_variable = { var = sofzh_merge_remaining value = 1 } }
 }
}
sofzh_random50_apply = {
 if = {
  limit = { sofzh_random50_window = yes }
  # Recover an old zero-merge attempt; a successful previous run remains locked.
  clr_global_flag = sofzh_random50_used
  set_variable = { var = sofzh_merge_count value = 0 }
  set_variable = { var = sofzh_merge_iterations value = 0 }
  sofzh_random50_count_countries = yes
  while_loop_effect = {
   limit = {
    check_variable = { var = sofzh_merge_remaining value = 50 compare = greater_than }
    check_variable = { var = sofzh_merge_iterations value = 512 compare = less_than }
    any_country = { sofzh_merge_donor = yes }
   }
   add_to_variable = { var = sofzh_merge_iterations value = 1 }
   random_country = {
    limit = { sofzh_merge_donor = yes }
    save_event_target_as = sofzh_random50_donor
    random_neighbor_country = {
     limit = { sofzh_merge_recipient = yes }
     save_event_target_as = sofzh_random50_recipient
     event_target:sofzh_random50_donor = {
      every_owned_state = { add_core_of = event_target:sofzh_random50_recipient }
     }
     annex_country = { target = event_target:sofzh_random50_donor transfer_troops = no }
     # Count only an annexation that really removed its donor.
     event_target:sofzh_random50_donor = {
      if = {
       limit = { exists = no }
       event_target:sofzh_random50_recipient = {
        set_country_flag = { flag = sofzh_ai_cooldown days = 90 }
       }
       ROOT = {
        add_to_variable = { var = sofzh_merge_count value = 1 }
        subtract_from_variable = { var = sofzh_merge_remaining value = 1 }
       }
      }
     }
    }
   }
  }
  # Recount the actual surviving countries; do not report the projected total.
  sofzh_random50_count_countries = yes
  if = {
   limit = { check_variable = { var = sofzh_merge_count value = 0 compare = greater_than } }
   set_global_flag = sofzh_random50_used
  }
  log = "[SOF random50] merges=[Root.sofzh_merge_count|0] remaining=[Root.sofzh_merge_remaining|0] attempts=[Root.sofzh_merge_iterations|0]"
  if = {
   limit = { check_variable = { var = sofzh_merge_remaining value = 50 compare = equals } }
   country_event = { id = sofzh_cabinet.2 }
  }
  else = { country_event = { id = sofzh_cabinet.3 } }
 }
