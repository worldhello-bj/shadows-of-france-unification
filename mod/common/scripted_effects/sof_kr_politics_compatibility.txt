# Preserve within-family ten-party proportions when a legacy effect sets four totals.
sof_kr_set_family_popularities = {
set_temp_variable = { sof_kr_old_family = party_popularity@totalist }
add_to_temp_variable = { sof_kr_old_family = party_popularity@syndicalist }
add_to_temp_variable = { sof_kr_old_family = party_popularity@radical_socialist }
if = { limit = { check_variable = { var = sof_kr_old_family value = 0 compare = greater_than } }
set_temp_variable = { sof_kr_new_totalist = party_popularity@totalist }
divide_temp_variable = { sof_kr_new_totalist = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_totalist = $COMMUNISM$ }
set_temp_variable = { sof_kr_new_syndicalist = party_popularity@syndicalist }
divide_temp_variable = { sof_kr_new_syndicalist = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_syndicalist = $COMMUNISM$ }
set_temp_variable = { sof_kr_new_radical_socialist = party_popularity@radical_socialist }
divide_temp_variable = { sof_kr_new_radical_socialist = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_radical_socialist = $COMMUNISM$ }
}
else = {
set_temp_variable = { sof_kr_new_totalist = 0 }
set_temp_variable = { sof_kr_new_syndicalist = 0 }
set_temp_variable = { sof_kr_new_radical_socialist = $COMMUNISM$ }
}
set_temp_variable = { sof_kr_old_family = party_popularity@social_democrat }
add_to_temp_variable = { sof_kr_old_family = party_popularity@social_liberal }
add_to_temp_variable = { sof_kr_old_family = party_popularity@market_liberal }
add_to_temp_variable = { sof_kr_old_family = party_popularity@social_conservative }
if = { limit = { check_variable = { var = sof_kr_old_family value = 0 compare = greater_than } }
set_temp_variable = { sof_kr_new_social_democrat = party_popularity@social_democrat }
divide_temp_variable = { sof_kr_new_social_democrat = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_social_democrat = $DEMOCRATIC$ }
set_temp_variable = { sof_kr_new_social_liberal = party_popularity@social_liberal }
divide_temp_variable = { sof_kr_new_social_liberal = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_social_liberal = $DEMOCRATIC$ }
set_temp_variable = { sof_kr_new_market_liberal = party_popularity@market_liberal }
divide_temp_variable = { sof_kr_new_market_liberal = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_market_liberal = $DEMOCRATIC$ }
set_temp_variable = { sof_kr_new_social_conservative = party_popularity@social_conservative }
divide_temp_variable = { sof_kr_new_social_conservative = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_social_conservative = $DEMOCRATIC$ }
}
else = {
set_temp_variable = { sof_kr_new_social_democrat = 0 }
set_temp_variable = { sof_kr_new_social_liberal = $DEMOCRATIC$ }
set_temp_variable = { sof_kr_new_market_liberal = 0 }
set_temp_variable = { sof_kr_new_social_conservative = 0 }
}
set_temp_variable = { sof_kr_old_family = party_popularity@authoritarian_democrat }
add_to_temp_variable = { sof_kr_old_family = party_popularity@paternal_autocrat }
if = { limit = { check_variable = { var = sof_kr_old_family value = 0 compare = greater_than } }
set_temp_variable = { sof_kr_new_authoritarian_democrat = party_popularity@authoritarian_democrat }
divide_temp_variable = { sof_kr_new_authoritarian_democrat = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_authoritarian_democrat = $NEUTRALITY$ }
set_temp_variable = { sof_kr_new_paternal_autocrat = party_popularity@paternal_autocrat }
divide_temp_variable = { sof_kr_new_paternal_autocrat = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_paternal_autocrat = $NEUTRALITY$ }
}
else = {
set_temp_variable = { sof_kr_new_authoritarian_democrat = $NEUTRALITY$ }
set_temp_variable = { sof_kr_new_paternal_autocrat = 0 }
}
set_temp_variable = { sof_kr_old_family = party_popularity@national_populist }
if = { limit = { check_variable = { var = sof_kr_old_family value = 0 compare = greater_than } }
set_temp_variable = { sof_kr_new_national_populist = party_popularity@national_populist }
divide_temp_variable = { sof_kr_new_national_populist = sof_kr_old_family }
multiply_temp_variable = { sof_kr_new_national_populist = $FASCISM$ }
}
else = {
set_temp_variable = { sof_kr_new_national_populist = $FASCISM$ }
}
set_popularities = {
totalist = sof_kr_new_totalist
syndicalist = sof_kr_new_syndicalist
radical_socialist = sof_kr_new_radical_socialist
social_democrat = sof_kr_new_social_democrat
social_liberal = sof_kr_new_social_liberal
market_liberal = sof_kr_new_market_liberal
social_conservative = sof_kr_new_social_conservative
authoritarian_democrat = sof_kr_new_authoritarian_democrat
paternal_autocrat = sof_kr_new_paternal_autocrat
national_populist = sof_kr_new_national_populist
}
}
