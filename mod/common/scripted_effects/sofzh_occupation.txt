# These effects run in STATE scope, ROOT is the occupier.
sofzh_occupation_clear = {
 remove_dynamic_modifier = { modifier = sofzh_local_underground }
 remove_dynamic_modifier = { modifier = sofzh_local_guerrillas }
 remove_dynamic_modifier = { modifier = sofzh_local_insurgency }
 remove_dynamic_modifier = { modifier = sofzh_local_reconstruction }
 remove_dynamic_modifier = { modifier = sofzh_local_accord }
 clr_state_flag = sofzh_occupation_registered
 clr_state_flag = sofzh_occupation_recent
 clr_state_flag = sofzh_registration_pending
 clr_state_flag = sofzh_governance_action
 clr_state_flag = sofzh_local_sweep
 clr_state_flag = sofzh_rebuild_cooldown
 clr_state_flag = sofzh_accord_cooldown
 clr_state_flag = sofzh_sweep_cooldown
 clr_state_flag = sofzh_foreign_weapons_cooldown
}
sofzh_occupation_register = {
 set_state_flag = sofzh_occupation_registered
 set_state_flag = { flag = sofzh_occupation_recent days = 180 }
 if = { limit = { has_resistance = no } start_resistance = yes }
 if = { limit = { resistance < 15 } set_resistance = 15 }
}
sofzh_occupation_refresh = {
 # Exactly one resistance band; temporary relief modifiers may coexist.
 remove_dynamic_modifier = { modifier = sofzh_local_underground }
 remove_dynamic_modifier = { modifier = sofzh_local_guerrillas }
 remove_dynamic_modifier = { modifier = sofzh_local_insurgency }
 if = { limit = { NOT = { resistance < 60 } }
  add_dynamic_modifier = { modifier = sofzh_local_insurgency }
 }
 else_if = { limit = { NOT = { resistance < 35 } }
  add_dynamic_modifier = { modifier = sofzh_local_guerrillas }
 }
 else = { add_dynamic_modifier = { modifier = sofzh_local_underground } }
}
sofzh_occupation_attack = {
 # Called only for successful monthly rolls; never creates field divisions.
 if = { limit = { NOT = { resistance < 60 } }
  ROOT = { add_manpower = -150 add_equipment_to_stockpile = { type = infantry_equipment amount = -120 } }
 }
 else = { ROOT = { add_manpower = -75 add_equipment_to_stockpile = { type = infantry_equipment amount = -60 } } }
 add_compliance = -1
 add_resistance = 2
 if = { limit = { infrastructure > 0 } damage_building = { type = infrastructure damage = 0.3 } }
 if = { limit = { arms_factory > 0 } damage_building = { type = arms_factory damage = 0.25 } }
 sofzh_occupation_refresh = yes
 if = { limit = { ROOT = { is_ai = no NOT = { has_country_flag = sofzh_guerrilla_report_cooldown } } }
  save_event_target_as = sofzh_guerrilla_report_state
  ROOT = {
   set_country_flag = { flag = sofzh_guerrilla_report_cooldown days = 90 }
   country_event = { id = sofzh_occupation.1 days = 1 }
  }
 }
}
sofzh_occupation_monthly = {
 every_controlled_state = {
  limit = { sofzh_unification_french_state = yes }
  if = { limit = { sofzh_occupation_state = yes }
   if = { limit = { NOT = { has_state_flag = sofzh_occupation_registered } } sofzh_occupation_register = yes }
   sofzh_occupation_refresh = yes
   if = { limit = { NOT = { has_state_flag = sofzh_local_sweep } }
    if = { limit = { NOT = { resistance < 60 } } random_list = { 40 = { sofzh_occupation_attack = yes } 60 = { } } }
    else_if = { limit = { NOT = { resistance < 35 } } random_list = { 20 = { sofzh_occupation_attack = yes } 80 = { } } }
   }
  }
  else = { sofzh_occupation_clear = yes }
 }
}
