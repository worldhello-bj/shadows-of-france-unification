  set_variable = { sofzh_chart_social = 0 }
  set_variable = { sofzh_chart_revolutionary = 0 }

  set_variable = { sofzh_chart_total = 0 }

  add_to_variable = { sofzh_chart_total = sofzh_chart_communism_pop }

  else_if = { limit = { sofzh_chart_party_corporatist = yes } set_variable = { sofzh_chart_fascism_frame = 9 } }
  set_variable = { sofzh_chart_fascism_pop = party_popularity_100@fascism }
  clamp_variable = { var = sofzh_chart_fascism_pop min = 0 max = 100 }

   multiply_variable = { sofzh_chart_communism_pop = 100 }
   if = { limit = { check_variable = { var = sofzh_chart_communism_frame value = 4 compare = equals } } set_variable = { sofzh_chart_revolutionary = sofzh_chart_communism_pop } }

  }

 }
}
