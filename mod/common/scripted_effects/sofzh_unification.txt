    if = {
        limit = { is_ai = yes }
        if = {
            limit = { sofzh_ai_campaign_ready = yes NOT = { has_country_flag = sofzh_ai_selected } }
            random_neighbor_country = {
                limit = {
                    sofzh_ai_safe_land_target = yes
                    NOT = { ROOT = { has_wargoal_against = PREV } }
                }
                ROOT = { create_wargoal = { type = annex_everything target = PREV expire = 730 } }
            }
        }
    }
    else = {
        every_neighbor_country = {
            limit = { sofzh_unification_new_land_target = yes }
            ROOT = { create_wargoal = { type = annex_everything target = PREV expire = 730 } }
        }
    }
}
