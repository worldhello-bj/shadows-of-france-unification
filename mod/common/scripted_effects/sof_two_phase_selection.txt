sof_tp_clear_selection = { clr_country_flag = sof_tp_has_selected clr_country_flag = sof_tp_selected_unaligned clear_variable = sof_tp_selected_country clear_variable = sof_tp_selected_leader clear_variable = sof_tp_selected_index clear_array = sof_tp_selected_country_list }
sof_tp_refresh_selection = { if = { limit = { has_country_flag = sof_tp_has_selected NOT = { sof_tp_has_valid_selection = yes } } sof_tp_clear_selection = yes } }
sof_tp_select_PRS_effect = {
 sof_tp_clear_selection = yes
 if = { limit = { PRS = { sof_tp_country = yes } }
  set_country_flag = sof_tp_has_selected set_variable = { sof_tp_selected_country = PRS } set_variable = { sof_tp_selected_index = 1 } add_to_array = { array = sof_tp_selected_country_list value = PRS }
  if = { limit = { PRS = { is_in_faction = no } } set_country_flag = sof_tp_selected_unaligned }
  else = { every_country = { limit = { is_faction_leader = yes is_in_faction_with = PRS } ROOT = { set_variable = { sof_tp_selected_leader = PREV } } } }
 }
}
